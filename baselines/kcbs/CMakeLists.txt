cmake_minimum_required(VERSION 3.5)
project(kcbs_bench)
# K-CBS (IMRCLab fork of OMPL, commit a307727 pinned by db-CBS) + db-CBS's main_kcbs driver.
find_package(Boost 1.58 REQUIRED COMPONENTS program_options)
find_package(Eigen3 REQUIRED)
find_package(PkgConfig)
pkg_check_modules(YAML REQUIRED yaml-cpp)
find_package(fcl REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(kcbs EXCLUDE_FROM_ALL)
include_directories(${EIGEN3_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/kcbs/src ${CMAKE_BINARY_DIR}/kcbs/src)
add_executable(main_kcbs src/main_kcbs.cpp src/robots.cpp)
target_link_libraries(main_kcbs ${Boost_LIBRARIES} ${YAML_LIBRARIES} ompl fcl)
